#include<stdio.h>
#include<stdlib.h>
#include<string.h>
struct book{
int book_id;
char title[50];
char author[50];
float price;
char status[10];
};
struct book*b;
int  n;

/*create book rewards*/
void create_book(){
    printf("Enter number of books: ");
    scanf("%d",&n);
    b=(struct book*)malloc(n*sizeof(struct book));
    for(int i=0;i<n;i++){
        printf("Enter book id: ");
        scanf("%d",&b[i].book_id);
        printf("Enter title: ");
        scanf("%s",b[i].title);
        printf("Enter author: ");
        scanf("%s",b[i].author);
        printf("Enter price: ");
        scanf("%f",&b[i].price);
        strcpy(b[i].status,"available");
    }
}
void display()
{int i;
printf("\n----available books----\n");
for(i=0;i<n;i++)
{
    if(strcmp(b[i].status,"available")==0)
    {
        printf("Book id: %d\n",b[i].book_id);
        printf("Title: %s\n",b[i].title);
        printf("Author: %s\n",b[i].author);
        printf("Price: %.2f\n",b[i].price);
        printf("Status: %s\n",b[i].status);
        printf("\n");
    }
}
}
void search()
{
    int id;
    printf("Enter book id to search: ");
    scanf("%d",&id);
    for(int i=0;i<n;i++)
    {
        if(b[i].book_id==id)
        {
            printf("Book id: %d\n",b[i].book_id);
            printf("Title: %s\n",b[i].title);
            printf("Author: %s\n",b[i].author);
            printf("Price: %.2f\n",b[i].price);
            printf("Status: %s\n",b[i].status);
            return;
        }
    }
    printf("Book not found.\n");
}

void issue_book()
{
    int id;
    printf("Enter book id to issue: ");
    scanf("%d",&id);
    for(int i=0;i<n;i++)
    {
        if(b[i].book_id==id)
        {
            if(strcmp(b[i].status,"available")==0)
            {
                strcpy(b[i].status,"issued");
                printf("Book issued successfully.\n");
                return;
            }
            else
            {
                printf("Book is already issued.\n");
                return;
            }
        }
    }
    printf("Book not found.\n");
}

void return_book()
{
    int id;
    printf("Enter book id to return: ");
    scanf("%d",&id);
    for(int i=0;i<n;i++)
    {
        if(b[i].book_id==id)
        {
            if(strcmp(b[i].status,"issued")==0)
            {
                strcpy(b[i].status,"available");
                printf("Book returned successfully.\n");
                return;
            }
            else
            {
                printf("Book is not issued.\n");
                return;
            }
        }
    }
    printf("Book not found.\n");
}

int main()
{
    int choice;
    do
    {
        printf("\n----Library Menu----\n");
        printf("1. Create book records\n");
        printf("2. Display available books\n");
        printf("3. Search for a book\n");
        printf("4. Issue a book\n");
        printf("5. Return a book\n");
        printf("6. Exit\n");
        printf("Enter your choice: ");
        scanf("%d",&choice);
        switch(choice)
        {
            case 1:
                create_book();
                break;
            case 2:
                display();
                break;
            case 3:
                search();
                break;
            case 4:
                issue_book();
                break;
            case 5:
                return_book();
                break;
            case 6:
                printf("Thankyou:\n");
                free(b);
                break;
            
            default:
                printf("Invalid choice.\n");
        }
    }while(choice!=6);
    return 0;
}
